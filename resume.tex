\newcommand{\resumeItem}[2]{
  \item{
    \color{black}\textbf{#1}\color{black}{: #2 \vspace{-4pt}}
  }
}

% ATS-friendly layout: \item[] avoids a stray bullet on its own line in the
% extracted text. Convention for the arguments:
%   #1 = role / institution (bold, left)   #2 = dates (right)
%   #3 = company + location / degree (italic, left)   #4 = optional (italic, right)
\newcommand{\resumeSubheading}[4]{
  \vspace{-1pt}\item[]
    \begin{tabular*}{0.97\textwidth}{l@{\extracolsep{\fill}}r}
      \textbf{#1} & #2 \\
      \textit{#3} & \textit{ #4} \\
    \end{tabular*}\vspace{-8pt}
}
